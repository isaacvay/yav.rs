//! Declarative Spring Motion & Scroll Reveal Card Component.
//! Generated via `yav add motion`.

use yav::prelude::*;

#[derive(Clone, Default)]
pub struct MotionCardProps {
    pub title: String,
    pub subtitle: String,
    pub children: Vec<Node>,
}

#[component]
pub fn MotionCard(props: MotionCardProps) -> Node {
    rsx! {
        <div
            class="plate p-6 space-y-4 cursor-pointer select-none will-change-transform"
            yav:reveal="is-in"
            yav:hover="scale:1.02, y:-4"
            yav:tap="scale:0.97"
            yav:spring="320, 18, 1"
        >
            <div class="font-mono text-xs uppercase tracking-wider text-oxide">
                {props.subtitle}
            </div>
            <h3 class="text-xl font-medium">
                {props.title}
            </h3>
            <div class="text-sm opacity-80">
                {props.children}
            </div>
        </div>
    }
}
