use yav::prelude::*;

#[derive(Clone, Default)]
pub struct DataRow {
    pub id: String,
    pub name: String,
    pub email: String,
    pub role: String,
    pub status: String,
}

#[component]
pub fn DataTable(title: String, rows: Vec<DataRow>) -> Node {
    let rendered_rows = rows.into_iter().map(|row| {
        let status_class = match row.status.as_str() {
            "Active" => "bg-emerald-500/10 text-emerald-500",
            "Pending" => "bg-amber-500/10 text-amber-500",
            _ => "bg-zinc-500/10 text-zinc-400",
        };

        rsx! {
            <tr class="border-b border-border hover:bg-muted/50 transition duration-150">
                <td class="py-3.5 px-4 text-sm font-medium">{row.name}</td>
                <td class="py-3.5 px-4 text-sm text-muted-foreground">{row.email}</td>
                <td class="py-3.5 px-4 text-sm">{row.role}</td>
                <td class="py-3.5 px-4 text-sm">
                    <span class={format!("px-2.5 py-1 text-xs font-semibold rounded-full {}", status_class)}>
                        {row.status}
                    </span>
                </td>
                <td class="py-3.5 px-4 text-sm text-right">
                    <a href={format!("/users/{}", row.id)} class="text-primary hover:underline text-xs font-semibold">
                        Gérer
                    </a>
                </td>
            </tr>
        }
    }).collect::<Vec<Node>>();

    rsx! {
        <div class="border border-border rounded-xl bg-card overflow-hidden shadow-sm">
            <div class="p-4 border-b border-border flex items-center justify-between">
                <h3 class="font-bold text-lg">{title}</h3>
                <span class="text-xs text-muted-foreground">Affichage des éléments récents</span>
            </div>
            <div class="overflow-x-auto">
                <table class="w-full text-left border-collapse">
                    <thead>
                        <tr class="border-b border-border bg-muted/30 text-xs uppercase text-muted-foreground">
                            <th class="py-3 px-4 font-semibold">Nom</th>
                            <th class="py-3 px-4 font-semibold">Email</th>
                            <th class="py-3 px-4 font-semibold">Rôle</th>
                            <th class="py-3 px-4 font-semibold">Statut</th>
                            <th class="py-3 px-4 font-semibold text-right">Actions</th>
                        </tr>
                    </thead>
                    <tbody>
                        {rendered_rows}
                    </tbody>
                </table>
            </div>
        </div>
    }
}
